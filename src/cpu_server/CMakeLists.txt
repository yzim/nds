find_package(PkgConfig REQUIRED)
pkg_check_modules(IBVERBS QUIET IMPORTED_TARGET libibverbs)

if(IBVERBS_FOUND)
    add_library(nds_verbs_backend STATIC backend/verbs/backend.cc)
    target_include_directories(nds_verbs_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(nds_verbs_backend PUBLIC nds_protocol nds_rdma_path_mtu PkgConfig::IBVERBS)
    target_compile_options(nds_verbs_backend PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_library(nds_cpu_transport STATIC transport/connection.cc)
    target_include_directories(nds_cpu_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(nds_cpu_transport PUBLIC nds_verbs_backend nds_peer_exchange)
    target_compile_options(nds_cpu_transport PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_library(nds_cpu_storage_protocol STATIC protocol/storage.cc)
    target_include_directories(nds_cpu_storage_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(nds_cpu_storage_protocol PUBLIC nds_cpu_transport nds_protocol)
    target_compile_options(nds_cpu_storage_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_executable(nds_verbs_server application/main.cc)
    target_link_libraries(nds_verbs_server PRIVATE CLI11::CLI11 nds_logging nds_cpu_storage_protocol)
    target_compile_options(nds_verbs_server PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    message(WARNING "libibverbs was not found; nds_verbs_server will not be built")
endif()
