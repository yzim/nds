find_package(PkgConfig REQUIRED)
pkg_check_modules(IBVERBS QUIET IMPORTED_TARGET libibverbs)

if(IBVERBS_FOUND)
    add_library(nds_cpu_storage_transport STATIC transport/storage_execution.cc)
    target_include_directories(nds_cpu_storage_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(nds_cpu_storage_transport PUBLIC nds_wire nds_logging PkgConfig::IBVERBS)
    target_compile_options(nds_cpu_storage_transport PRIVATE -Wall -Wextra -Wpedantic -Werror)

    add_executable(nds_verbs_server server.cc)
    target_link_libraries(nds_verbs_server PRIVATE CLI11::CLI11 nds_wire nds_rdma_path_mtu nds_peer_exchange nds_logging nds_cpu_storage_transport PkgConfig::IBVERBS)
    target_compile_options(nds_verbs_server PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    message(WARNING "libibverbs was not found; nds_verbs_server will not be built")
endif()
